const { Client, GatewayIntentBits } = require('discord.js'); const express = require('express'); const cors = require('cors'); // 1. Initialize Discord Bot Client with necessary intents const client = new Client({ intents: [ GatewayIntentBits.Guilds, GatewayIntentBits.GuildMembers, GatewayIntentBits.GuildPresences // Required to fetch online/idle/dnd/offline status ] }); // 2. Initialize Express Server for the Website API const app = express(); app.use(cors()); // Allows your website to fetch data from this server // --- CONFIGURATION --- const GUILD_ID = '1541430228771934218'; // Your Discord Server (Guild) ID // Your team categories and Discord User IDs const RAW_TEAM_CONFIG = { Founders: ["844809549222379561", "884810425315766353"], "IC Developer": ["876487460861919342", "1277162992923967519"], "DC Developer": ["1277162992923967519"], "Staff Team": [ "987346026476232714", "1223594753606680587", "853259849408053278", "877574020726403132" ], "Visa Managers": [ "1019946313154035744", "831766608770105354", "845448088076419093", ], }; // Keep a member in every configured category, including shared developer roles. const TEAM_CONFIG = RAW_TEAM_CONFIG; // 3. API Endpoint that your website will fetch data from app.get('/api/team', async (req, res) => { try { const guild = await client.guilds.fetch(GUILD_ID); const formattedRoles = []; for (const [category, userIds] of Object.entries(TEAM_CONFIG)) { const categoryMembers = []; for (const id of userIds) { try { // Fetch only this specific member if they aren't already cached const member = guild.members.cache.get(id) || await guild.members.fetch(id); if (member) { const presence = member.presence || guild.presences.cache.get(id); categoryMembers.push({ displayName: member.displayName, username: `@${member.user.username}`, avatar: member.user.displayAvatarURL({ extension: 'png', size: 128 }), status: presence?.status || 'offline', badge: member.roles.highest.name }); } } catch (error) { // User is not in the server anymore if (error.code === 10007) { console.log(`User ${id} is not in the guild.`); } else { console.error(`Failed to fetch member ${id}:`, error); } } } formattedRoles.push({ heading: category, members: categoryMembers }); } res.json({ roles: formattedRoles }); } catch (error) { console.error('Error handling /api/team request:', error); res.status(500).json({ error: 'Failed to fetch team data', roles: [] }); } }); // --- UptimeRobot Root Route --- app.get('/', (req, res) => { res.status(200).send('Thamizhan Roleplay API & Bot is UP!'); }); // Bot Ready Event client.once('ready', () => { console.log(`✅ Bot logged in successfully as ${client.user.tag}`); }); // Start Express Web Server with 0.0.0.0 binding for Render const PORT = process.env.PORT || 3000; app.listen(PORT, '0.0.0.0', () => { console.log(`🌐 Website Team API server is running on port ${PORT}`); }); // Log into Discord client.login(process.env.TOKEN);
